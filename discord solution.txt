(() => {
    delete window.$;
    let wpRequire = webpackChunkdiscord_app.push([[Symbol()], {}, r => r]);
    webpackChunkdiscord_app.pop();

    let ApplicationStreamingStore = Object.values(wpRequire.c).find(x => x?.exports?.A?.__proto__?.getStreamerActiveStreamMetadata)?.exports?.A;
    let RunningGameStore = Object.values(wpRequire.c).find(x => x?.exports?.Ay?.getRunningGames)?.exports?.Ay;
    let QuestsStore = Object.values(wpRequire.c).find(x => x?.exports?.A?.__proto__?.getQuest)?.exports?.A;
    let ChannelStore = Object.values(wpRequire.c).find(x => x?.exports?.A?.__proto__?.getAllThreadsForParent)?.exports?.A;
    let GuildChannelStore = Object.values(wpRequire.c).find(x => x?.exports?.Ay?.getSFWDefaultChannel)?.exports?.Ay;
    let FluxDispatcher = Object.values(wpRequire.c).find(x => x?.exports?.h?.__proto__?.flushWaitQueue)?.exports?.h;
    let api = Object.values(wpRequire.c).find(x => x?.exports?.Bo?.get)?.exports?.Bo;

    if (!QuestsStore) {
        console.error("QuestsStore не найден. Перезагрузи Discord, открой вкладку Quests и подожди 1 минуту.");
        return;
    }

    const supportedTasks = ["WATCH_VIDEO", "PLAY_ON_DESKTOP", "STREAM_ON_DESKTOP", "PLAY_ACTIVITY", "WATCH_VIDEO_ON_MOBILE"];

    function resolveApplicationId(quest, taskConfig, taskName) {
        const c = quest.config || {};
        const candidates = [
            c.application?.id,
            c.applicationId,
            c.assets?.applicationId,
            c.assets?.application_id,
            c.messages?.applicationId,
            taskConfig?.tasks?.[taskName]?.application?.id,
            taskConfig?.tasks?.[taskName]?.applicationId,
            taskConfig?.tasks?.[taskName]?.applications?.[0]?.id,
            taskConfig?.tasks?.[taskName]?.applications?.[0],
        ];
        for (const v of candidates) {
            if (v != null && v !== "") return String(v);
        }
        return null;
    }

    function resolveApplicationName(quest) {
        const c = quest.config || {};
        return (
            c.application?.name ||
            c.messages?.applicationName ||
            c.messages?.questName ||
            c.assets?.applicationName ||
            "Unknown"
        );
    }

    // Угадываем имя процесса, если executables пустой
    function guessExeName(appData) {
        const bases = [];
        if (appData?.name) bases.push(appData.name);
        if (Array.isArray(appData?.aliases)) bases.push(...appData.aliases);
        if (Array.isArray(appData?.executables)) {
            for (const e of appData.executables) {
                if (e?.name) bases.push(e.name.replace(/^>/, ""));
            }
        }
        for (const b of bases) {
            if (!b) continue;
            // Убираем всё кроме букв/цифр, добавляем .exe
            const cleaned = String(b).replace(/[^a-zA-Z0-9]/g, "");
            if (cleaned.length >= 2) return cleaned + ".exe";
        }
        return "game.exe";
    }

    let quests = [...QuestsStore.quests.values()].filter(x => 
        x.userStatus?.enrolledAt && 
        !x.userStatus?.completedAt && 
        new Date(x.config.expiresAt).getTime() > Date.now() && 
        supportedTasks.find(y => Object.keys((x.config.taskConfig ?? x.config.taskConfigV2).tasks).includes(y))
    );

    let isApp = typeof DiscordNative !== "undefined";

    if (quests.length === 0) {
        console.log("Нет незавершённых квестов! Убедись, что принял квест и подождал загрузки.");
        return;
    }

    let doJob = function() {
        const quest = quests.pop();
        if (!quest) return;

        const questName = quest.config?.messages?.questName || resolveApplicationName(quest);
        const taskConfig = quest.config.taskConfig ?? quest.config.taskConfigV2;

        if (!taskConfig || !taskConfig.tasks) {
            console.warn(`Пропуск квеста "${questName}": отсутствует taskConfig`);
            doJob();
            return;
        }

        // joinOperator "or" — пробуем задачи по порядку приоритета
        // (видео проще всего, потом игра/стрим/активити)
        const availableTasks = supportedTasks.filter(x => taskConfig.tasks[x] != null);
        if (availableTasks.length === 0) {
            console.warn(`Пропуск квеста "${questName}": нет поддерживаемой задачи`);
            doJob();
            return;
        }
        const taskName = availableTasks[0];

        const applicationId = resolveApplicationId(quest, taskConfig, taskName);
        const applicationName = resolveApplicationName(quest);
        const pid = Math.floor(Math.random() * 30000) + 1000;

        const secondsNeeded = taskConfig.tasks[taskName].target;
        let secondsDone = quest.userStatus?.progress?.[taskName]?.value ?? 0;

        if (taskName === "WATCH_VIDEO" || taskName === "WATCH_VIDEO_ON_MOBILE") {
            const maxFuture = 10, speed = 7, interval = 1;
            const enrolledAt = new Date(quest.userStatus.enrolledAt).getTime();
            let completed = false;
            let fn = async () => {            
                while (true) {
                    const maxAllowed = Math.floor((Date.now() - enrolledAt)/1000) + maxFuture;
                    const diff = maxAllowed - secondsDone;
                    const timestamp = secondsDone + speed;
                    if (diff >= speed) {
                        const res = await api.post({url: `/quests/${quest.id}/video-progress`, body: {timestamp: Math.min(secondsNeeded, timestamp + Math.random())}});
                        completed = res.body.completed_at != null;
                        secondsDone = Math.min(secondsNeeded, timestamp);
                        console.log(`Прогресс видео: ${secondsDone}/${secondsNeeded}`);
                    }
                    
                    if (timestamp >= secondsNeeded) {
                        break;
                    }
                    await new Promise(resolve => setTimeout(resolve, interval * 1000));
                }
                if (!completed) {
                    await api.post({url: `/quests/${quest.id}/video-progress`, body: {timestamp: secondsNeeded}});
                }
                console.log("Квест завершён!");
                doJob();
            };
            fn();
            console.log(`Имитация просмотра видео для ${questName}.`);
        } else if (taskName === "PLAY_ON_DESKTOP") {
            if (!isApp) {
                console.log("Этот тип работает только в десктопном приложении Discord для", questName);
                doJob();
                return;
            }
            if (!applicationId) {
                console.warn(`Пропуск квеста "${questName}": не удалось найти application.id для PLAY_ON_DESKTOP`);
                doJob();
                return;
            }
            api.get({url: `/applications/public?application_ids=${applicationId}`}).then(res => {
                const appData = res.body?.[0];
                if (!appData) {
                    console.warn(`Не удалось получить данные приложения для ${questName}, пропуск.`);
                    doJob();
                    return;
                }

                // Фолбэк: если executables нет — угадываем имя процесса
                let exeName;
                let hasRealExec = Array.isArray(appData.executables) && appData.executables.length > 0;
                if (hasRealExec) {
                    exeName = appData.executables.find(x => x.os === "win32")?.name?.replace(/^>/, "") 
                           ?? appData.name.replace(/[\/\\:*?"<>|]/g, "");
                } else {
                    exeName = guessExeName(appData);
                    console.warn(`У "${questName}" нет executables, пробуем угадать имя процесса: ${exeName}`);
                }

                if (!exeName || exeName.length < 3) {
                    console.warn(`Не удалось определить имя процесса для "${questName}", пропуск.`);
                    doJob();
                    return;
                }

                const processName = exeName.replace(/\.exe$/i, "");
                const fakeGame = {
                    cmdLine: `C:\\Program Files\\${appData.name || questName}\\${exeName}`,
                    exeName,
                    exePath: `c:/program files/${(appData.name || questName).toLowerCase()}/${exeName}`,
                    hidden: false,
                    isLauncher: false,
                    id: applicationId,
                    name: appData.name || questName,
                    pid: pid,
                    pidPath: [pid],
                    processName,
                    start: Date.now(),
                };
                const realGames = RunningGameStore.getRunningGames();
                const fakeGames = [fakeGame];
                const realGetRunningGames = RunningGameStore.getRunningGames;
                const realGetGameForPID = RunningGameStore.getGameForPID;
                RunningGameStore.getRunningGames = () => fakeGames;
                RunningGameStore.getGameForPID = (pid) => fakeGames.find(x => x.pid === pid);
                FluxDispatcher.dispatch({type: "RUNNING_GAMES_CHANGE", removed: realGames, added: [fakeGame], games: fakeGames});
                
                let fn = data => {
                    let progress = quest.config.configVersion === 1 ? data.userStatus.streamProgressSeconds : Math.floor(data.userStatus.progress.PLAY_ON_DESKTOP.value);
                    console.log(`Прогресс квеста: ${progress}/${secondsNeeded}`);
                    
                    if (progress >= secondsNeeded) {
                        console.log("Квест завершён!");
                        
                        RunningGameStore.getRunningGames = realGetRunningGames;
                        RunningGameStore.getGameForPID = realGetGameForPID;
                        FluxDispatcher.dispatch({type: "RUNNING_GAMES_CHANGE", removed: [fakeGame], added: [], games: []});
                        FluxDispatcher.unsubscribe("QUESTS_SEND_HEARTBEAT_SUCCESS", fn);
                        
                        doJob();
                    }
                };
                FluxDispatcher.subscribe("QUESTS_SEND_HEARTBEAT_SUCCESS", fn);
                
                console.log(`Имитация игры ${applicationName} (процесс ${exeName}). Осталось ~${Math.ceil((secondsNeeded - secondsDone) / 60)} мин.`);
                console.log("Если прогресс не растёт — Discord не принимает фейковый процесс. Тогда квест нужно выполнять реально.");
            }).catch(err => {
                console.error(`Ошибка при получении данных приложения для ${questName}:`, err);
                doJob();
            });
        } else if (taskName === "STREAM_ON_DESKTOP") {
            if (!isApp) {
                console.log("Этот тип работает только в десктопном приложении Discord для", questName);
                doJob();
                return;
            }
            if (!applicationId) {
                console.warn(`Пропуск квеста "${questName}": не удалось найти application.id для STREAM_ON_DESKTOP`);
                doJob();
                return;
            }
            let realFunc = ApplicationStreamingStore.getStreamerActiveStreamMetadata;
            ApplicationStreamingStore.getStreamerActiveStreamMetadata = () => ({
                id: applicationId,
                pid,
                sourceName: null
            });
            
            let fn = data => {
                let progress = quest.config.configVersion === 1 ? data.userStatus.streamProgressSeconds : Math.floor(data.userStatus.progress.STREAM_ON_DESKTOP.value);
                console.log(`Прогресс квеста: ${progress}/${secondsNeeded}`);
                
                if (progress >= secondsNeeded) {
                    console.log("Квест завершён!");
                    
                    ApplicationStreamingStore.getStreamerActiveStreamMetadata = realFunc;
                    FluxDispatcher.unsubscribe("QUESTS_SEND_HEARTBEAT_SUCCESS", fn);
                    
                    doJob();
                }
            };
            FluxDispatcher.subscribe("QUESTS_SEND_HEARTBEAT_SUCCESS", fn);
            
            console.log(`Имитация стрима ${applicationName}. Стрими любое окно в VC ~${Math.ceil((secondsNeeded - secondsDone) / 60)} мин.`);
            console.log("Нужен хотя бы 1 человек в VC!");
        } else if (taskName === "PLAY_ACTIVITY") {
            const channelId = ChannelStore.getSortedPrivateChannels()[0]?.id ?? Object.values(GuildChannelStore.getAllGuilds()).find(x => x != null && x.VOCAL.length > 0)?.VOCAL[0]?.channel.id;
            if (!channelId) {
                console.warn(`Пропуск квеста "${questName}": не найден голосовой канал для PLAY_ACTIVITY (зайди в любой VC).`);
                doJob();
                return;
            }
            const streamKey = `call:${channelId}:1`;
            
            let fn = async () => {
                console.log("Выполнение квеста", questName);
                
                while (true) {
                    const res = await api.post({url: `/quests/${quest.id}/heartbeat`, body: {stream_key: streamKey, terminal: false}});
                    const progress = res.body.progress.PLAY_ACTIVITY.value;
                    console.log(`Прогресс квеста: ${progress}/${secondsNeeded}`);
                    
                    await new Promise(resolve => setTimeout(resolve, 20 * 1000));
                    
                    if (progress >= secondsNeeded) {
                        await api.post({url: `/quests/${quest.id}/heartbeat`, body: {stream_key: streamKey, terminal: true}});
                        break;
                    }
                }
                
                console.log("Квест завершён!");
                doJob();
            };
            fn();
        }
    };
    doJob();
})();
